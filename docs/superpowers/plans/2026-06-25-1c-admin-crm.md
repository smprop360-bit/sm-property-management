```typescript
'use client'

import { useState, useEffect, useMemo } from 'react'
import { createClient } from '@/lib/supabase/client'
import type { Lead, LeadStatus } from '@/lib/types/lead'

export function useLeads() {
  const [leads, setLeads] = useState<Lead[]>([])
  const [loading, setLoading] = useState(true)
  const [searchQuery, setSearchQuery] = useState('')
  const [filterType, setFilterType] = useState('')
  const [filterStatus, setFilterStatus] = useState('')

  useEffect(() => {
    const supabase = createClient()
    supabase
      .from('leads')
      .select('*')
      .order('created_at', { ascending: false })
      .then(({ data }) => {
        setLeads(data ?? [])
        setLoading(false)
      })
  }, [])

  const filteredLeads = useMemo(() => {
    return leads.filter(lead => {
      const q = searchQuery.toLowerCase()
      const matchesSearch =
        !q ||
        lead.full_name.toLowerCase().includes(q) ||
        lead.phone.includes(q)
      const matchesType = !filterType || lead.property_type === filterType
      const matchesStatus = !filterStatus || lead.status === filterStatus
      return matchesSearch && matchesType && matchesStatus
    })
  }, [leads, searchQuery, filterType, filterStatus])

  async function updateStatus(id: number, status: LeadStatus): Promise<void> {
    const supabase = createClient()
    await supabase.from('leads').update({ status }).eq('id', id)
    setLeads(prev => prev.map(l => (l.id === id ? { ...l, status } : l)))
  }

  async function saveNotes(id: number, notes: string): Promise<void> {
    const supabase = createClient()
    await supabase.from('leads').update({ notes }).eq('id', id)
    setLeads(prev => prev.map(l => (l.id === id ? { ...l, notes } : l)))
  }

  return {
    leads: filteredLeads,
    loading,
    searchQuery,
    setSearchQuery,
    filterType,
    setFilterType,
    filterStatus,
    setFilterStatus,
    updateStatus,
    saveNotes,
  }
}
```

```typescript
'use client'

import { useState } from 'react'
import type { Lead } from '@/lib/types/lead'

export function useLeadDrawer() {
  const [selectedLead, setSelectedLead] = useState<Lead | null>(null)
  const [notesValue, setNotesValue] = useState('')

  const open = selectedLead !== null

  function openDrawer(lead: Lead) {
    setSelectedLead(lead)
    setNotesValue(lead.notes ?? '')
  }

  function closeDrawer() {
    setSelectedLead(null)
    setNotesValue('')
  }

  return { selectedLead, notesValue, setNotesValue, open, openDrawer, closeDrawer }
}
```

```tsx
'use client'

import { useEffect, useState } from 'react'
import {
  BarChart,
  Bar,
  XAxis,
  YAxis,
  Tooltip,
  ResponsiveContainer,
  CartesianGrid,
} from 'recharts'
import { createClient } from '@/lib/supabase/client'

interface DayCount {
  day: string
  count: number
}

export function LeadsChart() {
  const [data, setData] = useState<DayCount[]>([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    async function fetchData() {
      const supabase = createClient()
      const thirtyDaysAgo = new Date()
      thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30)

      const { data: rows } = await supabase
        .from('leads')
        .select('created_at')
        .gte('created_at', thirtyDaysAgo.toISOString())

      const counts: Record<string, number> = {}
      const today = new Date()
      for (let i = 29; i >= 0; i--) {
        const d = new Date(today)
        d.setDate(d.getDate() - i)
        const key = d.toISOString().slice(0, 10)
        counts[key] = 0
      }
      for (const row of rows ?? []) {
        const key = (row.created_at as string).slice(0, 10)
        if (key in counts) counts[key]++
      }

      setData(
        Object.entries(counts).map(([day, count]) => ({
          day: day.slice(5),
          count,
        }))
      )
      setLoading(false)
    }
    fetchData()
  }, [])

  if (loading) {
    return (
      <div className="bg-white rounded-xl p-6 shadow-sm">
        <div className="h-[220px] bg-gray-50 animate-pulse rounded" />
      </div>
    )
  }

  return (
    <div className="bg-white rounded-xl p-6 shadow-sm">
      <h3 className="text-base font-semibold text-brand-navy mb-4">לידים — 30 יום אחרונים</h3>
```

```tsx
import { useEffect, useState } from 'react'
import { UserPlus, Users, Building2 } from 'lucide-react'
import { createClient } from '@/lib/supabase/client'
import { AdminStatsCard } from '@/components/admin/AdminStatsCard'
import { LeadsChart } from '@/components/admin/LeadsChart'
import { RecentLeadsTable } from '@/components/admin/RecentLeadsTable'

export default function AdminDashboardPage() {
  const [todayCount, setTodayCount] = useState(0)
  const [weekCount, setWeekCount] = useState(0)
  const [activeProperties, setActiveProperties] = useState(0)
  const [statsLoading, setStatsLoading] = useState(true)

  useEffect(() => {
    async function fetchStats() {
      const supabase = createClient()
      const today = new Date()
      today.setHours(0, 0, 0, 0)
      const sevenDaysAgo = new Date()
      sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7)
      sevenDaysAgo.setHours(0, 0, 0, 0)

      const [{ count: todayC }, { count: weekC }, { count: propC }] = await Promise.all([
        supabase
          .from('leads')
          .select('*', { count: 'exact', head: true })
          .gte('created_at', today.toISOString()),
        supabase
          .from('leads')
          .select('*', { count: 'exact', head: true })
          .gte('created_at', sevenDaysAgo.toISOString()),
        supabase
          .from('properties')
          .select('*', { count: 'exact', head: true })
          .eq('status', 'active'),
      ])

      setTodayCount(todayC ?? 0)
      setWeekCount(weekC ?? 0)
      setActiveProperties(propC ?? 0)
      setStatsLoading(false)
    }
    fetchStats()
  }, [])

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold text-brand-navy">דשבורד</h1>
      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">

          value={todayCount}
          icon={UserPlus}
          loading={statsLoading}
        />

          label="לידים השבוע"
          value={weekCount}
          icon={Users}
          loading={statsLoading}
        />
        <AdminStatsCard
          label="נכסים פעילים"
          value={activeProperties}
          icon={Building2}
          loading={statsLoading}
        />
      </div>
```

```typescript
'use client'

import { useState, useEffect, useMemo } from 'react'
import { createClient } from '@/lib/supabase/client'
import type { Property } from '@/lib/types/property'

export function useProperties() {
  const [properties, setProperties] = useState<Property[]>([])
  const [loading, setLoading] = useState(true)
  const [searchQuery, setSearchQuery] = useState('')
  const [filterType, setFilterType] = useState('')

  useEffect(() => {
    createClient()
      .from('properties')
      .select('*')
      .order('created_at', { ascending: false })
      .then(({ data }) => {
        setProperties(data ?? [])
        setLoading(false)
      })
  }, [])

  const filteredProperties = useMemo(() => {
    return properties.filter(p => {
      const q = searchQuery.toLowerCase()
      const matchesSearch =
        !q ||
        p.title.toLowerCase().includes(q) ||
        p.city.toLowerCase().includes(q)
      const matchesType = !filterType || p.type === filterType
      return matchesSearch && matchesType
    })
  }, [properties, searchQuery, filterType])

  async function deleteProperty(id: string): Promise<void> {
    const supabase = createClient()
    await supabase.from('properties').delete().eq('id', id)
    setProperties(prev => prev.filter(p => p.id !== id))
  }

  return {
    properties: filteredProperties,
    loading,
    searchQuery,
    setSearchQuery,
    filterType,
    setFilterType,
    deleteProperty,
  }
}
```

```typescript
'use client'

import { useState, ChangeEvent, FormEvent } from 'react'
import { useRouter } from 'next/navigation'
import { createClient } from '@/lib/supabase/client'
import type { PropertyType, DealType, PropertyStatus } from '@/lib/types/property'

export interface PropertyFormValues {
  title: string
  type: string
  deal_type: string
  status: string
  city: string
  address: string
  area: string
  size_sqm: string
  floor: string
  price: string
  price_visible: boolean
  price_text: string
  description: string
  images: string[]
}

type FormErrors = Partial<Record<keyof PropertyFormValues, string>>

const DEFAULT_VALUES: PropertyFormValues = {
  title: '',
  type: '',
  deal_type: '',
  status: 'active',
  city: '',
  address: '',
  area: '',
  size_sqm: '',
  floor: '',
  price: '',
  price_visible: true,
  price_text: '',
  description: '',
  images: [],
}

function generateSlug(title: string): string {
  const base =
    title
      .trim()
      .toLowerCase()
      .replace(/\s+/g, '-')
      .replace(/[^a-z0-9-]/g, '')
      .replace(/--+/g, '-')
      .slice(0, 40) || 'property'
  return `${base}-${Date.now().toString(36)}`
}

async function uploadImages(
  supabase: ReturnType<typeof createClient>,
  propertyId: string,
  files: File[]
): Promise<string[]> {
  const urls: string[] = []
  for (const file of files) {
    const ext = file.name.split('.').pop() ?? 'jpg'
    const path = `${propertyId}/${Date.now()}.${ext}`
    const { data, error } = await supabase.storage
      .from('property-images')
      .upload(path, file, { upsert: false })
    if (!error && data) {
      const {
        data: { publicUrl },
      } = supabase.storage.from('property-images').getPublicUrl(data.path)
      urls.push(publicUrl)
    }
  }
  return urls
}

export function usePropertyForm(
  initialValues?: Partial<PropertyFormValues>,
  propertyId?: string
) {
  const router = useRouter()
  const [values, setValues] = useState<PropertyFormValues>({
    ...DEFAULT_VALUES,
    ...initialValues,
  })
  const [errors, setErrors] = useState<FormErrors>({})
  const [loading, setLoading] = useState(false)
  const [submitError, setSubmitError] = useState<string | null>(null)
  const [existingImages, setExistingImages] = useState<string[]>(
    initialValues?.images ?? []
  )
  const [imageFiles, setImageFiles] = useState<File[]>([])

  function handleChange(e: ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) {
    const { name, value } = e.target
    setValues(prev => ({ ...prev, [name]: value }))
    if (errors[name as keyof PropertyFormValues]) {
      setErrors(prev => ({ ...prev, [name]: undefined }))
    }
  }

  function handleSelectChange(name: keyof PropertyFormValues, value: string) {
    setValues(prev => ({ ...prev, [name]: value }))
    if (errors[name]) {
      setErrors(prev => ({ ...prev, [name]: undefined }))
    }
  }

  function handleBoolChange(name: keyof PropertyFormValues, value: boolean) {
    setValues(prev => ({ ...prev, [name]: value }))
  }

  function handleImageAdd(files: File[]) {
    setImageFiles(prev => [...prev, ...files])
  }

  function handleImageRemove(url: string) {
    setExistingImages(prev => prev.filter(u => u !== url))
  }

  function handleNewImageRemove(index: number) {
    setImageFiles(prev => prev.filter((_, i) => i !== index))
  }

  function validate(): boolean {
    const newErrors: FormErrors = {}
    if (!values.title.trim()) newErrors.title = 'כותרת הוא שדה חובה'
    if (!values.type) newErrors.type = 'סוג נכס הוא שדה חובה'
    if (!values.deal_type) newErrors.deal_type = 'סוג עסקה הוא שדה חובה'
    if (!values.city.trim()) newErrors.city = 'עיר הוא שדה חובה'
    setErrors(newErrors)
    return Object.keys(newErrors).length === 0
  }

  async function handleSubmit(e: FormEvent) {
    e.preventDefault()
    if (!validate()) return

    setLoading(true)
    setSubmitError(null)
    const supabase = createClient()

    try {
      const sharedFields = {
        title: values.title.trim(),
        type: values.type as PropertyType,
        deal_type: values.deal_type as DealType,
        status: values.status as PropertyStatus,
        city: values.city.trim(),
        address: values.address.trim() || null,
        area: values.area.trim() || null,
        size_sqm: values.size_sqm ? parseInt(values.size_sqm, 10) : 0,
        floor: values.floor.trim() || null,
        price: values.price ? parseInt(values.price, 10) : null,
        price_visible: values.price_visible,
        price_text: values.price_text.trim() || null,
        description: values.description.trim() || null,
      }

      if (propertyId) {
        const newUrls = await uploadImages(supabase, propertyId, imageFiles)
        const allImages = [...existingImages, ...newUrls]
        const { error } = await supabase
          .from('properties')
          .update({ ...sharedFields, images: allImages })
          .eq('id', propertyId)
        if (error) throw error
      } else {
        const { data, error } = await supabase
          .from('properties')
          .insert({ ...sharedFields, slug: generateSlug(values.title), features: [], images: [] })
          .select('id')
          .single()
        if (error) throw error
        if (imageFiles.length > 0 && data) {
          const newUrls = await uploadImages(supabase, data.id, imageFiles)
          await supabase.from('properties').update({ images: newUrls }).eq('id', data.id)
        }
      }

      router.push('/admin/properties')
    } catch {
      setSubmitError('שגיאה בשמירת הנכס. אנא נסה שוב.')
    } finally {
      setLoading(false)
    }
  }

  return {
    values,
    errors,
    loading,
    submitError,
    existingImages,
    imageFiles,
    handleChange,
    handleSelectChange,
    handleBoolChange,
    handleImageAdd,
    handleImageRemove,
    handleNewImageRemove,
    handleSubmit,
  }
}
```

```tsx
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            <div className="space-y-1.5">
              <Label htmlFor="size_sqm">גודל (מ"ר)</Label>
              <Input id="size_sqm" name="size_sqm" type="number" min="0" value={values.size_sqm} onChange={onChange} placeholder="120" />
            </div>
            <div className="space-y-1.5">
              <Label htmlFor="floor">קומה</Label>
              <Input id="floor" name="floor" value={values.floor} onChange={onChange} placeholder="3" />
            </div>
          </div>
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            <div className="space-y-1.5">
              <Label htmlFor="price">מחיר (₪)</Label>
              <Input id="price" name="price" type="number" min="0" value={values.price} onChange={onChange} placeholder="15000" />
            </div>
            <div className="space-y-1.5">
              <Label>הצג מחיר</Label>
              <div className="flex items-center gap-2 h-10">
                <input
                  type="checkbox"
                  id="price_visible"
                  checked={values.price_visible}
                  onChange={e => onBoolChange('price_visible', e.target.checked)}
                  className="w-4 h-4 accent-brand-gold"
                />
                <Label htmlFor="price_visible" className="cursor-pointer font-normal">
                  כן, הצג מחיר
                </Label>
              </div>
            </div>
          </div>
          {!values.price_visible && (
            <div className="space-y-1.5">
              <Label htmlFor="price_text">טקסט מחיר</Label>
              <Input
                id="price_text"
                name="price_text"
                value={values.price_text}
                onChange={onChange}
                placeholder="מחיר לפי פנייה"
              />
            </div>
          )}
          <div className="space-y-1.5">
            <Label htmlFor="description">תיאור</Label>
            <Textarea
              id="description"
              name="description"
              value={values.description}
              onChange={onChange}
              rows={4}
              placeholder="תיאור מפורט של הנכס..."
            />
          </div>
        </section>

        {/* Section 4 — תמונות */}
        <section className="bg-white rounded-xl p-6 shadow-sm space-y-4">
          <h2 className="text-base font-semibold text-brand-navy border-b pb-3">תמונות</h2>
          {(existingImages.length > 0 || imageFiles.length > 0) && (
            <div className="flex flex-wrap gap-3">
              {existingImages.map(url => (
                <div key={url} className="relative">
                  <img src={url} alt="" className="w-20 h-20 rounded-lg object-cover" />
                  <button
                    type="button"
                    onClick={() => onImageRemove(url)}
                    className="absolute -top-1.5 -end-1.5 w-5 h-5 bg-red-500 text-white rounded-full flex items-center justify-center hover:bg-red-600"
                    aria-label="הסר תמונה"
                  >
                    <X size={10} />
                  </button>
                </div>
              ))}
              {imageFiles.map((file, i) => (
                <div key={i} className="relative">
                  <img
                    src={URL.createObjectURL(file)}
                    alt=""
                    className="w-20 h-20 rounded-lg object-cover"
                  />
                  <button
                    type="button"
                    onClick={() => onNewImageRemove(i)}
                    className="absolute -top-1.5 -end-1.5 w-5 h-5 bg-red-500 text-white rounded-full flex items-center justify-center hover:bg-red-600"
                    aria-label="הסר תמונה"
                  >
                    <X size={10} />
                  </button>
                </div>
              ))}
            </div>
          )}
          <label className="flex flex-col items-center justify-center gap-2 border-2 border-dashed border-gray-200 rounded-xl p-8 cursor-pointer hover:border-brand-gold hover:bg-brand-gold/5 transition-colors">
            <Upload size={24} className="text-gray-400" />
            <span className="text-sm text-gray-500">לחץ להוספת תמונות</span>
            <input
              type="file"
              accept="image/*"
              multiple
              className="hidden"
              onChange={e => {
                if (e.target.files) onImageAdd(Array.from(e.target.files))
              }}
            />
          </label>
        </section>

        {/* Actions */}
        <div className="flex gap-3">
          <button
            type="submit"
            disabled={loading}
            className="h-11 px-8 rounded-lg bg-brand-gold text-white font-semibold text-sm hover:bg-brand-gold/90 disabled:opacity-50"
          >
            {loading ? 'שומר...' : isEdit ? 'שמור שינויים' : 'הוסף נכס'}
          </button>
          <button
            type="button"
            onClick={() => router.push('/admin/properties')}
            className="h-11 px-8 rounded-lg border border-input text-sm font-medium hover:bg-gray-50"
          >
            ביטול
          </button>
        </div>

        {submitError && <p className="text-red-500 text-sm">{submitError}</p>}
      </div>
    </form>
  )
}
```

- [ ] **Step 3: Create edit property page**

Create `src/app/(admin)/admin/properties/[id]/edit/page.tsx`:

```tsx
'use client'

import { useEffect, useState } from 'react'
import { useParams } from 'next/navigation'
import { createClient } from '@/lib/supabase/client'
import type { Property } from '@/lib/types/property'
import { usePropertyForm } from '@/hooks/admin/usePropertyForm'
import { PropertyForm } from '@/components/admin/PropertyForm'

export default function EditPropertyPage() {
  const params = useParams()
  const id = params.id as string
  const [property, setProperty] = useState<Property | null>(null)
  const [notFound, setNotFound] = useState(false)

  useEffect(() => {
    createClient()
      .from('properties')
      .select('*')
      .eq('id', id)
      .single()
      .then(({ data, error }) => {
        if (error || !data) {
          setNotFound(true)
        } else {
          setProperty(data)
        }
      })
  }, [id])

  if (notFound) {
    return <p className="text-red-500">נכס לא נמצא</p>
  }
  if (!property) {
    return (
      <div className="space-y-4">
        <div className="h-8 w-48 bg-gray-100 animate-pulse rounded" />
        <div className="h-64 bg-white rounded-xl shadow-sm animate-pulse" />
      </div>
    )
  }

  return <EditPropertyForm property={property} />
}

function EditPropertyForm({ property }: { property: Property }) {
  const form = usePropertyForm(
    {
      title: property.title,
      type: property.type,
      deal_type: property.deal_type,
      status: property.status,
      city: property.city,
      address: property.address ?? '',
      area: property.area ?? '',
      size_sqm: property.size_sqm?.toString() ?? '',
      floor: property.floor ?? '',
      price: property.price?.toString() ?? '',
      price_visible: property.price_visible,
      price_text: property.price_text ?? '',
      description: property.description ?? '',
      images: property.images,
    },
    property.id
  )

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold text-brand-navy">עריכת נכס: {property.title}</h1>
      <PropertyForm
        values={form.values}
        errors={form.errors}
        loading={form.loading}
        submitError={form.submitError}
        existingImages={form.existingImages}
        imageFiles={form.imageFiles}
        onChange={form.handleChange}
        onSelectChange={form.handleSelectChange}
        onBoolChange={form.handleBoolChange}
        onImageAdd={form.handleImageAdd}
        onImageRemove={form.handleImageRemove}
        onNewImageRemove={form.handleNewImageRemove}
        onSubmit={form.handleSubmit}
        isEdit
      />
    </div>
  )
}
```
